number = int(input("Введите натуральное значение: "))
digitCopy = 0
digitalSquare = 0
numberLength = 1

if number % 10 == number:
    print(number)
else:
    while number % (10 ** numberLength) != number:
        numberLength += 1
    
    for i in range(numberLength - 1, 0, -1):
        digitalSquare += number // (10 ** i)
        digitCopy = number % (10 ** i)
        number -= digitCopy * (10 ** i)
print(digitalSquare)
