summaryeven = 0
summarypositive = 0
count = 0
countpositive = 0

n = -1

while n != 0:
    n = float(input("Введите значение: "))
    if n != 0:
        count += 1
    
    if n % 2 == 0:
        summaryeven += n
    
    if n > 0:
        countpositive += 1
        summarypositive += n

if summarypositive > 0:    
    print(f"Количество введеных значений: {count}, сумма всех четных значений: {summaryeven}, среднее арифметическое положительных значений: {summarypositive / countpositive}")
else:
    print(f"Количество введеных значений: {count}, сумма всех четных значений: {summaryeven}, среднее арифметическое положительных значений не вычислено в связи с отсутствием положительных значений")
