using System;
					
public class Program
{
	public static void Main()
	{
	int input;
		do
		{
			Console.WriteLine("Выберите действие: 1 - вычисление факториала; 2 - вычисление НОД; 3 - выход");
			input = Convert.ToInt32(Console.ReadLine());
			if ((input > 0) & (input < 4))
			{
				switch(input)
				{
					case 1:
						Console.WriteLine("Введите число для вычисления его факториала: ");
						double inputFactorial = Convert.ToDouble(Console.ReadLine());
						double factorial = 1;
						for (int i = 1; i <= inputFactorial; i++)
						{
							factorial *= i;
						}
						break;
						
					case 2:
                        double inputNOD1, inputNOD2, r;
                        Console.WriteLine("Введите первое число для вычисления НОД: ");
                        inputNOD1 = Convert.ToDouble(Console.ReadLine());
                        Console.WriteLine("Введите второе число для вычисления НОД: ");
                        inputNOD2 = Convert.ToDouble(Console.ReadLine());
                        
                        if (inputNOD1 < 0)
                        {
                            inputNOD1 *= -1;
                        }
                        
                        if (inputNOD2 < 0)
                        {
                            inputNOD2 *= -1;
                        }
                    
                        if (inputNOD1 < inputNOD2)
                        {
                            double temp = inputNOD1;
                            inputNOD1 = inputNOD2;
                            inputNOD2 = temp;
                        }
                    
                        while (inputNOD2 != 0)
                        {
                            r = inputNOD1 % inputNOD2;
                            inputNOD1 = inputNOD2;
                            inputNOD2 = r;
                        }
                    
                        Console.WriteLine($"НОД является число {inputNOD1}");
                        break;
    						
					case 3:
						Console.WriteLine("Программа остановлена");
						break;
				}
			}
			else
			{
				Console.WriteLine("Ошибка");
			}
		} 
		while (input != 3);
	}
}

using System;

class Program
{
    static void Main()
    {
        Console.Write("Введите до какого числа вывести ряд Фибоначчи: ");
        int N = Convert.ToInt32(Console.ReadLine());

        int previous = 0;  
        int current = 1;  

        if (N < 1)
        {
            Console.WriteLine("Чисел Фибоначчи, не превосходящих введенное число, нет.");
            return;
        }

        for (int i = 1; current <= N; i++)
        {
            Console.WriteLine($"{i}: {current}");

            int next = previous + current;
            previous = current;
            current = next;
        }
    }
}
